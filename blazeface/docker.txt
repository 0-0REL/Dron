# Imagen ESPECÍFICA para Raspberry Pi 4 (armv7l)
FROM arm32v7/python:3.7-bullseye

# Configurar pip para usar piwheels (CRUCIAL)
RUN echo "[global]" > /etc/pip.conf && \
    echo "extra-index-url = https://www.piwheels.org/simple" >> /etc/pip.conf && \
    echo "trusted-host = www.piwheels.org" >> /etc/pip.conf

# Dependencias compatibles con ARM - NOMBRES CORRECTOS para Bullseye
RUN apt-get install -y \
    gfortran \
    libhdf5-dev \
    libc-ares-dev \
    libeigen3-dev \
    && rm -rf /var/lib/apt/lists/*

# Establecer el directorio de trabajo
WORKDIR /app

# Copiar requirements.txt primero
COPY requirements.txt .

# Instalar dependencias de Python COMPATIBLES con ARM
RUN pip install --no-cache-dir -r requirements.txt

# Copiar el script principal
COPY prueba.py .

# Copiar los archivos del modelo
COPY BlazeFaceDetection/ .
COPY models .

# Exponer puertos
EXPOSE 5000 5001

# Comando para ejecutar la aplicación
CMD ["python", "prueba.py"]
