\subsection{Dinámica}
El modelo dinámico del cuadricóptero es
\begin{gather}
	m\ddot{\mathbf{r}} = \mathbf{R}_{\phi,\theta,\psi}\begin{bmatrix}
	0\\0\\F\end{bmatrix} - \begin{bmatrix}0\\0\\mg\end{bmatrix} - \mathbf{F}_f\\
	M(\boldsymbol{\eta})\ddot{\boldsymbol{\eta}} = \boldsymbol{\tau} - C(\boldsymbol{\eta},\dot{\boldsymbol{\eta}})\dot{\boldsymbol{\eta}} - \boldsymbol{\tau}_f
\end{gather}
\begin{table}
	\centering
	\caption{Parámetros del dron}
	\label{tab:param_dron}
	\begin{tabular}{l l l}
		$m$ & 1.6 & kg \\
		$I_{xx}$ & 0.021915027 & kgm$^2$ \\
		$I_{yy} $ & 0.022124548 & kgm$^2$ \\
		$I_{zz}$ & 0.042349865 & kgm$^2$ \\
		$L$ & 0.23 & m \\
		$K_F$ & 6.11 $\times 10^{-8}$ & N/ms
	\end{tabular}
\end{table}
De \eqref{eq:mot2u} se obtiene la relación entre entradas de control y actuadores
\begin{equation}
	\label{eq:u2mot}
	\begin{bmatrix}
		\omega_1^2\\
		\omega_2^2\\
		\omega_3^3\\
		\omega_4^2
	\end{bmatrix}
	=
	\left[\begin{array}{cccc}
		\frac{1}{4\,K_F } & -\frac{1}{4\,K_F \,L}  & -\frac{1}{4\,K_F \,L}  & -\frac{1}{4\,K_M }\\
		\frac{1}{4\,K_F } & -\frac{1}{4\,K_F \,L}  & \frac{1}{4\,K_F \,L}  & \frac{1}{4\,K_M }\\
		\frac{1}{4\,K_F } & \frac{1}{4\,K_F \,L}  & \frac{1}{4\,K_F \,L}  & -\frac{1}{4\,K_M }\\
		\frac{1}{4\,K_F } & \frac{1}{4\,K_F \,L}  & -\frac{1}{4\,K_F \,L} & \frac{1}{4\,K_M }
	\end{array}\right]
	\begin{bmatrix}
		F\\
		\tau_x\\
		\tau_y\\
		\tau_z
	\end{bmatrix}
\end{equation}

\subsubsection{Linealización}
Mediante la serie de Taylor \eqref{eq:taylor} el modelo del cuadricóptero se puede expresar de forma lineal de la siguiente forma
\begin{equation}
	\label{eq:modelo_lineal}
\begin{bmatrix}
\dot{\mathbf{r}} \\ \ddot{\mathbf{r}} \\ \dot{\boldsymbol{\eta}} \\ \ddot{\boldsymbol{\eta}}
\end{bmatrix}
=
\begin{bmatrix}
\mathit{0}_{3\times3} & I_{3\times3} & \mathit{0}_{3\times3} & \mathit{0}_{3\times3} \\
\mathit{0}_{3\times3} & \mathit{0}_{3\times3} & G(\psi_d) & \mathit{0}_{3\times3} \\
\mathit{0}_{3\times3} & \mathit{0}_{3\times3} & \mathit{0}_{3\times3} & I_{3\times3} \\
\mathit{0}_{3\times3} & \mathit{0}_{3\times3} & \mathit{0}_{3\times3} & \mathit{0}_{3\times3}
\end{bmatrix}
\begin{bmatrix}
\mathbf{r} \\ \dot{\mathbf{r}} \\ \boldsymbol{\eta} \\ \dot{\boldsymbol{\eta}}
\end{bmatrix}
+
\begin{bmatrix}
\mathit{0}_{5\times1} & \mathit{0}_{5\times3} \\
m^{-1} & \mathit{0}_{1\times3}\\
\mathit{0}_{3\times1} & \mathit{0}_{3\times3} \\
\mathit{0}_{3\times1} & I_d
\end{bmatrix}
\begin{bmatrix}
F \\ \boldsymbol{\tau}
\end{bmatrix}
\end{equation}

donde:
\[
\mathbf{r} = \begin{bmatrix}x \\ y \\ z\end{bmatrix}, \quad
\boldsymbol{\eta } = \begin{bmatrix}\phi \\ \theta \\ \psi\end{bmatrix}, \quad
\boldsymbol{\tau} = \begin{bmatrix}\tau_x \\ \tau_y \\ \tau_z\end{bmatrix}
\]
\[
\mathit{G}(\psi_d) = \begin{bmatrix}
0 & g & 0\\
-g & 0 & 0\\
0 & 0 & 0
\end{bmatrix}, \quad
I_d = \operatorname{diag}\left(I_{xx}^{-1}, I_{yy}^{-1}, I_{zz}^{-1}\right)
\]
\subsection{Algoritmos de control}
\subsubsection{PD}
El controlador se ajusta mediante las funciones de trasferencia, que se obtienen mediante \eqref{eq:ss2tf}, con
\[
\mathbf{C} = \begin{bmatrix}
	I_{3\times3} & \mathit{0}_{3\times3} & \mathit{0}_{3\times3} & \mathit{0}_{3\times3} \\
    \mathit{0}_{3\times3} & \mathit{0}_{3\times3} & I_{3\times3} & \mathit{0}_{3\times3}
\end{bmatrix}
\]
\[
\mathbf{D} = \mathit{0}_{6\times4}
\]
\begin{gather}
	G_x(s) = \frac{g}{I_{yy} s^4} \\
    G_y(s) = -\frac{g}{I_{xx} s^4} \\
    G_z(s) = \frac{1}{m s^2} \label{eq:tf_crtb_i} \\
    G_{\phi}(s) = \frac{1}{I_{xx} s^2} \\
    G_{\theta}(s) = \frac{1}{I_{yy} s^2} \\
    G_{\psi}(s) = \frac{1}{I_{zz} s^2} \label{eq:tf_crtb_f}
\end{gather}

Dado que no se puede medir los desplazamientos en $x$ y $y$, solo se realizan 4 de los 6 controladores (control de orientación y altura).
Dado que el cambio de la velocidad de los motores no es instantáneo, las ecuaciones \eqref{eq:tf_crtb_f}-\eqref{eq:tf_crtb_f} se multiplican por \eqref{eq:tf_mot}, que es la
función de trasferencia de los motores, para incluir este efecto.
\begin{eqnarray}
	\label{eq:tf_mot}
	\frac{K_m}{s+K_m}
\end{eqnarray}

Para ajustar los controladores se utiliza el método de colocación de las raíces, de modo que \eqref{eq:tf_crtb_i} a \eqref{eq:tf_crtb_f} se convierten en
\begin{gather}
	K_{Dz}\frac{(s+a_z)\frac{K_m}{m}}{s^2 (s+K_m)}\\
	K_{D\phi}\frac{(s+a_\phi)\frac{K_m}{I_{xx}}}{s^2 (s+K_m)}\\
	K_{D\theta}\frac{(s+a_\theta)\frac{K_m}{I_{yy}}}{s^2 (s+K_m)}\\
	K_{D\psi}\frac{(s+a_\psi)\frac{K_m}{I_{zz}}}{s^2 (s+K_m)}
\end{gather}
